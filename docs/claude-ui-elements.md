# Claude UI Elements: Design Document

## Overview

This document outlines the core UI elements required for displaying Claude's content JSON and rendering artifacts appropriately within the interface. The focus is on creating a responsive, accessible interface that accurately represents Claude's responses, including special artifact formats, while maintaining a clean and intuitive user experience.

## Current Understanding

Based on my observations of how Claude's UI functions, I understand that:

- Claude responses are delivered as structured JSON that includes text content and artifact specifications
- The UI needs to render both regular text responses and specialized artifact formats (code, markdown, HTML, SVG, Mermaid diagrams, React components)
- The interface should maintain conversation history and distinguish between user and assistant messages
- The UI needs to handle specialized rendering for different artifact types

## Artifact Type Renderers

### 1. Code Artifact Renderer

For rendering code snippets with syntax highlighting.

```javascript
class CodeArtifactRenderer {
  constructor(element, language) {
    this.element = element;
    this.language = language;
  }
  
  render(content) {
    // Remove any existing content
    this.element.innerHTML = '';
    
    // Create pre and code elements
    const pre = document.createElement('pre');
    const code = document.createElement('code');
    code.className = `language-${this.language}`;
    code.textContent = content;
    
    // Append elements
    pre.appendChild(code);
    this.element.appendChild(pre);
    
    // Apply syntax highlighting if Prism.js is available
    if (window.Prism) {
      Prism.highlightElement(code);
    }
    
    // Add copy button
    this.addCopyButton();
  }
  
  addCopyButton() {
    const copyButton = document.createElement('button');
    copyButton.className = 'copy-button';
    copyButton.textContent = 'Copy';
    copyButton.addEventListener('click', () => {
      const code = this.element.querySelector('code');
      navigator.clipboard.writeText(code.textContent)
        .then(() => {
          copyButton.textContent = 'Copied!';
          setTimeout(() => {
            copyButton.textContent = 'Copy';
          }, 2000);
        });
    });
    
    const header = this.element.closest('.artifact-container').querySelector('.artifact-header');
    header.appendChild(copyButton);
  }
}
```

class MarkdownArtifactRenderer {
  constructor(element) {
    this.element = element;
  }
  
  render(content) {
    // Remove any existing content
    this.element.innerHTML = '';
    
    // Render markdown if a library like marked.js is available
    if (window.marked) {

      this.element.textContent = content;

### 3. HTML Artifact Renderer

For rendering HTML content safely within an iframe or controlled environment.

```javascript
class HtmlArtifactRenderer {
  constructor(element) {
    this.element = element;
  }
  
  render(content) {
    // Create a sandboxed iframe for rendering HTML content
    const iframe = document.createElement('iframe');
    iframe.sandbox = 'allow-scripts';
    iframe.style.width = '100%';
    iframe.style.border = 'none';
    iframe.style.minHeight = '200px';
    
    // Attach iframe to the DOM
    this.element.innerHTML = '';
    this.element.appendChild(iframe);
    
    // Write content to the iframe
    const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
    iframeDoc.open();
    iframeDoc.write(content);
    iframeDoc.close();
    
    // Adjust iframe height to match content
    this.adjustIframeHeight(iframe);
  }
  
  adjustIframeHeight(iframe) {
    // Ensure iframe height matches content height
    iframe.onload = () => {
      const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
      const height = iframeDoc.body.scrollHeight;
      iframe.style.height = `${height + 20}px`;
    };
  }
}
```

### 4. SVG Artifact Renderer

For rendering SVG content with proper viewbox handling.

```javascript
class SvgArtifactRenderer {
  constructor(element) {
    this.element = element;
  }
  
  render(content) {
    // SVG content can be directly inserted into the DOM
    this.element.innerHTML = content;
    
    // Ensure SVG is responsive
    const svg = this.element.querySelector('svg');
    if (svg) {
      svg.style.width = '100%';
      svg.style.height = 'auto';
      
      // Ensure viewBox is set if not already
      if (!svg.getAttribute('viewBox') && 
          svg.getAttribute('width') && 
          svg.getAttribute('height')) {
        const width = svg.getAttribute('width');
        const height = svg.getAttribute('height');
        svg.setAttribute('viewBox', `0 0 ${width} ${height}`);
      }
    }
  }
}
```

### 6. React Component Renderer

For rendering React components with proper environment setup.

```javascript
class ReactArtifactRenderer {
  constructor(element) {
    this.element = element;
  }
  
  render(content) {
    // Create a container for React rendering
    const container = document.createElement('div');
    container.id = `react-container-${Date.now()}`;
    
    // Append container to the DOM
    this.element.innerHTML = '';
    this.element.appendChild(container);
    
    // Check if React and ReactDOM are available
    if (window.React && window.ReactDOM) {
      try {
        // Create a script element to evaluate the React component
        const script = document.createElement('script');
        script.type = 'text/babel';
        
        // Wrap content in a component definition if needed
        const wrappedContent = this.wrapContentIfNeeded(content);
        script.textContent = `
          ${wrappedContent}
          ReactDOM.render(
            React.createElement(Component),
            document.getElementById('${container.id}')
          );
        `;
        
        // Append script to the DOM
        document.body.appendChild(script);
        
        // Remove script after execution
        setTimeout(() => {
          document.body.removeChild(script);
        }, 100);
      } catch (error) {
        console.error('Error rendering React component:', error);
        container.textContent = 'Error rendering React component';
      }
    } else {
      console.warn('React libraries not loaded. Displaying raw content instead.');
      container.textContent = content;
    }
  }
  
  wrapContentIfNeeded(content) {
    // Check if content is already a component definition
    if (content.includes('function Component') || 
        content.includes('class Component') || 
        content.includes('const Component =')) {
      return content;
    }
    
    // Wrap content in a component definition
    return `const Component = () => {
      return (${content});
    }`;
  }
}
```

## Artifact Factory

A factory class to create the appropriate renderer based on artifact type.

```javascript
class ArtifactRendererFactory {
  static createRenderer(artifactType, element) {
    switch (artifactType) {
      case 'application/vnd.ant.code':
        return new CodeArtifactRenderer(element, 'javascript'); // Default language
      case 'text/markdown':
        return new MarkdownArtifactRenderer(element);
      case 'text/html':
        return new HtmlArtifactRenderer(element);
      case 'image/svg+xml':
        return new SvgArtifactRenderer(element);
      case 'application/vnd.ant.mermaid':
        return new MermaidArtifactRenderer(element);
      case 'application/vnd.ant.react':
        return new ReactArtifactRenderer(element);
      default:
        // Fallback to simple text display
        return {
          render: (content) => {
            element.textContent = content;
          }
        };
    }
  }
}
```

## JSON Parser and Message Renderer

The main controller for parsing Claude's JSON responses and rendering them appropriately.

```javascript
class ClaudeMessageRenderer {
  constructor(messageContainer) {
    this.messageContainer = messageContainer;
  }
  
  renderMessage(messageData) {
    // Create message element
    const messageElement = document.createElement('div');
    messageElement.className = messageData.role === 'user' ? 'user-message' : 'claude-message';
    
    // Render text content
    if (messageData.content) {
      const textElement = document.createElement('div');
      textElement.className = 'message-text';
      textElement.textContent = messageData.content;
      messageElement.appendChild(textElement);
    }
    
    // Render artifacts if present
    if (messageData.artifacts && messageData.artifacts.length > 0) {
      messageData.artifacts.forEach(artifact => {
        this.renderArtifact(messageElement, artifact);
      });
    }
    
    // Append message to container
    this.messageContainer.appendChild(messageElement);
    
    // Scroll to bottom
    this.messageContainer.scrollTop = this.messageContainer.scrollHeight;
  }
  
  renderArtifact(parentElement, artifact) {
    // Create artifact container
    const artifactContainer = document.createElement('div');
    artifactContainer.className = 'artifact-container';
    
    // Create artifact header
    const header = document.createElement('div');
    header.className = 'artifact-header';
    
    // Create artifact title
    const title = document.createElement('div');
    title.className = 'artifact-title';
    title.textContent = artifact.title || 'Artifact';
    header.appendChild(title);
    
    // Create artifact content container
    const content = document.createElement('div');
    content.className = 'artifact-content';
    
    // Create and use appropriate renderer
    const renderer = ArtifactRendererFactory.createRenderer(artifact.type, content);
    renderer.render(artifact.content);
    
    // Assemble artifact elements
    artifactContainer.appendChild(header);
    artifactContainer.appendChild(content);
    parentElement.appendChild(artifactContainer);
  }
  
  parseAndRenderResponse(responseJSON) {
    try {
      const responseData = JSON.parse(responseJSON);
      
      // Check if response is an array of messages
      if (Array.isArray(responseData)) {
        responseData.forEach(message => this.renderMessage(message));
      } else {
        // Assume it's a single message
        this.renderMessage(responseData);
      }
    } catch (error) {
      console.error('Error parsing Claude response:', error);
      
      // Create error message
      const errorMessage = document.createElement('div');
      errorMessage.className = 'claude-message error';
      errorMessage.textContent = 'Error rendering response';
      this.messageContainer.appendChild(errorMessage);
    }
  }
}
```

## Main Application

The main application that initializes the UI and handles interactions.

```javascript
class ClaudeUI {
  constructor(containerId) {
    this.container = document.getElementById(containerId);
    if (!this.container) {
      throw new Error(`Container element with ID "${containerId}" not found`);
    }
    
    this.messageContainer = null;
    this.messageRenderer = null;
    
    this.initialize();
  }
  
  initialize() {
    // Create message container
    this.messageContainer = document.createElement('div');
    this.messageContainer.className = 'message-container';
    this.container.appendChild(this.messageContainer);
    
    // Initialize message renderer
    this.messageRenderer = new ClaudeMessageRenderer(this.messageContainer);
    
    // Load required libraries (syntax highlighting, markdown parser, etc.)
    this.loadLibraries();
  }
  
  loadLibraries() {
    // Load syntax highlighting library (Prism.js)
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/prism/1.28.0/prism.min.js');
    this.loadStyle('https://cdnjs.cloudflare.com/ajax/libs/prism/1.28.0/themes/prism.min.css');
    
    // Load Marked.js for Markdown parsing
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/marked/4.2.12/marked.min.js');
    
    // Load Mermaid.js for diagram rendering
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/mermaid/9.1.7/mermaid.min.js');
    
    // Load React for component rendering
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/react/18.2.0/umd/react.production.min.js');
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/react-dom/18.2.0/umd/react-dom.production.min.js');
    this.loadScript('https://cdnjs.cloudflare.com/ajax/libs/babel-standalone/7.18.12/babel.min.js');
  }
  
  loadScript(url) {
    const script = document.createElement('script');
    script.src = url;
    document.head.appendChild(script);
  }
  
  loadStyle(url) {
    const link = document.createElement('link');
    link.rel = 'stylesheet';
    link.href = url;
    document.head.appendChild(link);
  }
  
  renderResponse(responseJSON) {
    this.messageRenderer.parseAndRenderResponse(responseJSON);
  }
}
```

            document.body.appendChild(script);
            
            // Remove script after execution
            setTimeout(() => {
              document.body.removeChild(script);
            }, 100);
          } catch (error) {
            console.error('Error rendering React component:', error);
            container.textContent = 'Error rendering React component';
          }
        } else {
          console.warn('React libraries not loaded. Displaying raw content instead.');
          container.textContent = content;
        }
      }
      
      wrapContentIfNeeded(content) {
        // Check if content is already a component definition
        if (content.includes('function Component') || 
            content.includes('class Component') || 
            content.includes('const Component =')) {
          return content;
        }
        
        // Wrap content in a component definition
        return `const Component = () => {
          return (${content});
        }`;
      }
    }

    class ArtifactRendererFactory {
      static createRenderer(artifactType, element) {
        switch (artifactType) {
          case 'application/vnd.ant.code':
            return new CodeArtifactRenderer(element, 'javascript'); // Default language
          case 'text/markdown':
            return new MarkdownArtifactRenderer(element);
          case 'text/html':
            return new HtmlArtifactRenderer(element);
          case 'image/svg+xml':
            return new SvgArtifactRenderer(element);
          case 'application/vnd.ant.mermaid':
            return new MermaidArtifactRenderer(element);
          case 'application/vnd.ant.react':
            return new ReactArtifactRenderer(element);
          default:
            // Fallback to simple text display
            return {
              render: (content) => {
                element.textContent = content;
              }
            };
        }
      }
    }
